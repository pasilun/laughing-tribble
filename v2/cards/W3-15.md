Do (code compiled and run by the spec owner on 2026-11-10 with svg2pdf 0.13.0; it gave an A3 page of 1190.55 × 841.89 pt with DejaVu Sans embedded and selectable text):
```rust
use std::sync::Arc;
use svg2pdf::usvg;

/// The PDF of a sheet SVG (one page, the sheet's paper size).
pub fn pdf(sheet: &crate::sheet::Sheet) -> Result<Vec<u8>, Diagnostic> {
    let mut db = usvg::fontdb::Database::new();
    db.load_font_data(include_bytes!("../assets/DejaVuSans.ttf").to_vec());
    db.set_sans_serif_family("DejaVu Sans");
    let opt = usvg::Options { fontdb: Arc::new(db), ..Default::default() };
    let tree = usvg::Tree::from_str(&sheet.svg, &opt).map_err(|e| e012(&e.to_string()))?;
    svg2pdf::to_pdf(
        &tree,
        svg2pdf::ConversionOptions::default(),
        // usvg reads "420mm" as px at 96 dpi; 96 here turns that back into true mm.
        svg2pdf::PageOptions { dpi: 96.0 },
    )
    .map_err(|e| e012(&e.to_string()))
}
```
`e012` is the existing helper in `png.rs` (make it `pub(crate)` if it is not). This `usvg` is svg2pdf's own re-export (0.45), separate from resvg's; do not try to share types between the two.

pub struct Store { root: PathBuf }

`src/lib.rs` (code verified by the spec owner against rmcp 3.9.0; keep the structure):

```rust
use rmcp::{
    ErrorData as McpError, ServerHandler,
    handler::server::{router::tool::ToolRouter, wrapper::Parameters},
    model::{CallToolResult, Implementation, ServerCapabilities, ServerConfig},
    tool, tool_handler, tool_router,
};

#[derive(Debug, serde::Deserialize, schemars::JsonSchema)]
pub struct CheckArgs {
    /// The plan: a file name under the server's root, without `.plan`.
    pub plan_id: String,
}

#[derive(Debug, Clone)]
pub struct PlanServer { store: Store, tool_router: ToolRouter<Self> }

#[tool_router(router = tool_router)]
impl PlanServer {
    pub fn new(store: Store) -> Self { Self { store, tool_router: Self::tool_router() } }

    #[tool(name = "check", description = CHECK_DESCRIPTION, annotations(read_only_hint = true))]
    pub async fn check(&self, Parameters(args): Parameters<CheckArgs>) -> Result<CallToolResult, McpError> {
        Ok(tool_result(self.check_value(&args.plan_id)))
    }
}

#[tool_handler(router = self.tool_router)]
impl ServerHandler for PlanServer {
    fn get_info(&self) -> ServerConfig {
        let mut info = ServerConfig::new(ServerCapabilities::builder().enable_tools().build());
        info.server_info = Implementation::new("planstudio", env!("CARGO_PKG_VERSION"));
        info.instructions = Some(INSTRUCTIONS.to_string());
        info
    }
}
```

Helpers you write in lib.rs:
- `fn check_value(&self, plan_id: &str) -> Result<serde_json::Value, Vec<Diagnostic>>`: `store.load` (Err(d) → Err(vec![d])), then `plan_core::load(&text)`; on `Err(ds)` return `Ok(json!({"plan_id", "resolved": false, "diagnostics": ds}))` (a plan with parse errors is still checkable); on Ok return `json!({"plan_id", "resolved": r.resolved.is_some(), "diagnostics": r.diagnostics})`.
- `fn tool_result(v: Result<Value, Vec<Diagnostic>>) -> CallToolResult`: `Ok(v)` → `CallToolResult::structured(v)`; `Err(ds)` → `CallToolResult::structured_error(json!({"diagnostics": ds}))`. Tool-level problems are never `Err(McpError)`.
- `pub const INSTRUCTIONS: &str` = `"Planstudio keeps floor plans as text (a small DSL of rooms, relations, measurements and openings). Every number comes from the kernel or a laser reading, never from pixels. Working order: check, then get (view dsl) to read the plan, then apply a DSL patch (dry_run first if unsure), then check again. Use render only when you need to see the layout; refer to elements by the ids printed on the render."`
- `pub const CHECK_DESCRIPTION: &str` = `"Check a plan. Returns every diagnostic (errors, warnings, info) and what to do next."`

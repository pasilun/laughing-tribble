Do: the first step of moving walls from centre lines to faces. A wall gets an optional `ref: 'left' | 'right'`: when present, `a → b` is that face of the wall (seen from a towards b) and the body lies on the other side; without it, `a → b` is the centre line as today, so every existing plan is unchanged. A new pure module `lib/plan/domain/faces.ts` turns a plan into its centre-line view (`centreLines`): each face-referenced wall is moved by half its thickness, each end is run to the centre line of the wall it meets (an L corner where stored ends coincide, or a T where the end lies on the other's stored line), and an opening's offset grows by the length its wall's a-end moved, so the opening stays where it is drawn. A plan with no face-referenced wall is returned as it is (the same object), so the view is safe to apply anywhere and applying it twice is the same as once. Nothing reads the view yet (F-02 does). The fixture is lgh 154 with every wall stored on a room face, from the spike of 2026-11-11 (`spikes/face-walls/lgh154-face.json`, copied unchanged).

Why: a laser measures between faces, and today every measurement must be converted to a centre line by hand, and a thickness change moves both faces (the spike: making the south wall of Sovrum 100 mm thinner changed the room from 3.887 to 3.980 m²). Steps 4 and 5 (measurements as a source, the measurement kernel) need faces to point at.

2. New file `lib/plan/domain/faces.ts`:
```ts
import type { Opening, Plan, Pt, Wall } from './types'

// Walls stored on a face (wall.ref) and the centre-line view the rest of the code reads while it still assumes a body
// centred on the stored line. The view is temporary: the code moves to one function for a wall's body, a few places
// at a time, and then the view is deleted (docs/v2-engine-plan.md, section 3).
//
// The view of a plan with no face-referenced wall is the plan itself, so it is safe to apply at any entry point, and
// applying it twice is the same as once (the view has no face-referenced walls).

const sub = (p: Pt, q: Pt): Pt => ({ x: p.x - q.x, y: p.y - q.y })
const norm = (p: Pt) => Math.hypot(p.x, p.y)
const near = (p: Pt, q: Pt) => norm(sub(p, q)) <= 1

function dirOf(w: Wall): Pt {
  const d = sub(w.b, w.a)
  const l = norm(d) || 1
  return { x: d.x / l, y: d.y / l }
}

// From the stored line to the centre line: a left face has its body on the right of a → b, and the other way round.
// The left normal of a direction (dx, dy) in plan coordinates (y down) is (dy, -dx).
export function centreShift(w: Wall): Pt {
  if (!w.ref) return { x: 0, y: 0 }
  const d = dirOf(w)
  const k = (w.ref === 'left' ? -1 : 1) * (w.thickness / 2)
  return { x: d.y * k, y: -d.x * k }
}

function meet(p: Pt, d: Pt, q: Pt, e: Pt): Pt | null {
  const den = d.x * e.y - d.y * e.x
  if (Math.abs(den) < 0.05) return null
  const t = ((q.x - p.x) * e.y - (q.y - p.y) * e.x) / den
  return { x: p.x + d.x * t, y: p.y + d.y * t }
}

// Whether p lies on the inside of w's stored line (a T-joint against it), within 1 mm.
function onLine(p: Pt, w: Wall): boolean {
  const d = dirOf(w)
  const v = sub(p, w.a)
  const t = v.x * d.x + v.y * d.y
  return Math.abs(v.x * d.y - v.y * d.x) <= 1 && t > 1 && t < norm(sub(w.b, w.a)) - 1
}

const round = (p: Pt): Pt => ({ x: Math.round(p.x), y: Math.round(p.y) })

// A wall's centre line in the view: moved by half its thickness, and each end run to the centre line of the wall it
// meets (an L corner, where stored ends coincide, or a T, where the end lies on the other's stored line).
function centreLine(w: Wall, walls: Wall[]): { a: Pt; b: Pt } {
  const s = centreShift(w)
  const d = dirOf(w)
  const end = (p: Pt): Pt => {
    const c = { x: p.x + s.x, y: p.y + s.y }
    for (const o of walls) {
      if (o.id === w.id) continue
      if (!near(o.a, p) && !near(o.b, p) && !onLine(p, o)) continue
      const so = centreShift(o)
      const m = meet(c, d, { x: o.a.x + so.x, y: o.a.y + so.y }, dirOf(o))
      if (m) return m
    }
    return c
  }
  return { a: round(end(w.a)), b: round(end(w.b)) }
}

export function hasFaceWalls(plan: Pick<Plan, 'walls'>): boolean {
  return plan.walls.some((w) => w.ref)
}

// The plan as the centre-line code reads it: every wall on its centre line, without ref. An opening is measured from
// its wall's a-end; where the view moved that end along the wall, the offset grows by the same length, so the opening
// stays where it is drawn.
export function centreLines<P extends Pick<Plan, 'walls'> & { openings?: Opening[] }>(plan: P): P {
  if (!hasFaceWalls(plan)) return plan
  const grow = new Map<string, number>()
  const walls = plan.walls.map((w): Wall => {
    const { a, b } = centreLine(w, plan.walls)
    const s = centreShift(w)
    const d = dirOf(w)
    const moved = sub({ x: w.a.x + s.x, y: w.a.y + s.y }, a)
    grow.set(w.id, moved.x * d.x + moved.y * d.y)
    const { ref: _ref, ...rest } = w
    void _ref
    return { ...rest, a, b }
  })
  const openings = plan.openings?.map((o) => (grow.get(o.wallId) ? { ...o, offset: Math.round(o.offset + grow.get(o.wallId)!) } : o))
  return { ...plan, walls, ...(openings ? { openings } : {}) }
}
```

import { test, expect } from '@playwright/test'
import { readFileSync } from 'fs'
import { centreLines, centreShift } from '@/lib/plan/domain/faces'
import { validatePlan } from '@/lib/plan/domain/ops'
import { roomAreas } from '@/lib/plan/domain/rooms'
import type { Plan, Wall } from '@/lib/plan/domain/types'

// Walls stored on a face (wall.ref) and the centre-line view the rest of the code reads. lgh154-face.json is lgh 154
// with every wall stored on a room face (the spike of 2026-11-11, spikes/face-walls/).

const face = () => JSON.parse(readFileSync('e2e/fixtures/faces/lgh154-face.json', 'utf8')) as Plan
// Room areas are read from the view, as every reader of the plan does while the view exists.
const area = (p: Plan, room: string) => Math.round(roomAreas(centreLines(p), 'fore').get(room)! / 1e3) / 1e3
const noPolygons = (p: Plan): Plan => ({ ...p, rooms: p.rooms.map((r) => ({ ...r, polygon: undefined })) })

test('a plan with no face-referenced wall is its own view', () => {
  for (const f of ['e2e/fixtures/flat/drawing-1999.json', 'e2e/fixtures/golden/reference.json']) {
    const p = JSON.parse(readFileSync(f, 'utf8')) as Plan
    expect(centreLines(p)).toBe(p)
  }
})

test('the view of lgh 154 on faces is a valid plan with the same room areas as the centre-line plan', () => {
  const v = centreLines(face())
  expect(validatePlan(v)).toEqual([])
  expect(v.walls.some((w) => 'ref' in w)).toBe(false)
  expect([area(v, 'Garderob'), area(v, 'Hall'), area(v, 'Sovrum'), area(v, 'Vardagsrum'), area(v, 'WC')]).toEqual([4.39, 1.224, 3.887, 12.389, 2.346])
  // Applying it twice is the same as once.
  expect(centreLines(v)).toBe(v)
})

test('a thinner face-referenced wall keeps its face, and the room keeps its area', () => {
  const p = noPolygons(face())
  const id = 'W:Sovrum.south:outer'
  const face0 = (q: Plan) => {
    const w = centreLines(q).walls.find((x) => x.id === id)!
    return { inner: w.a.y - w.thickness / 2, outer: w.a.y + w.thickness / 2 }
  }
  expect(face0(p)).toEqual({ inner: 4540, outer: 4840 })
  const thin: Plan = { ...p, walls: p.walls.map((w) => (w.id === id ? { ...w, thickness: 200 } : w)) }
  expect(face0(thin)).toEqual({ inner: 4540, outer: 4740 })
  expect(area(thin, 'Sovrum')).toBe(area(p, 'Sovrum'))
  // The same change on the centre-line plan moves both faces and changes the room.
  const centre = centreLines(p)
  const c2: Plan = { ...centre, walls: centre.walls.map((w) => (w.id === id ? { ...w, thickness: 200 } : w)) }
  const w2 = c2.walls.find((x) => x.id === id)!
  expect([w2.a.y - 100, w2.a.y + 100]).toEqual([4590, 4790])
  expect(area(c2, 'Sovrum')).toBeGreaterThan(area(p, 'Sovrum'))
})

test('a face-referenced wall meets a centre-line wall and a wall at 45° at one point', () => {
  const walls: Wall[] = [
    { id: 'north', a: { x: 0, y: 0 }, b: { x: 4000, y: 0 }, thickness: 300, ref: 'right', phase: 'existing' },
    { id: 'west', a: { x: 0, y: 0 }, b: { x: 0, y: 3000 }, thickness: 200, phase: 'existing' },
    { id: 'slant', a: { x: 4000, y: 0 }, b: { x: 6000, y: 2000 }, thickness: 300, ref: 'right', phase: 'existing' },
  ]
  const v = centreLines({ walls })
  const at = (id: string) => v.walls.find((w) => w.id === id)!
  // Body to the north: the centre line is 150 mm above the stored face.
  expect(centreShift(walls[0])).toEqual({ x: 0, y: -150 })
  expect(at('north').a).toEqual(at('west').a)
  expect(at('north').b).toEqual(at('slant').a)
})

test('an opening stays where it is drawn when the view moves its wall\'s a-end', () => {
  const walls: Wall[] = [
    { id: 'n', a: { x: 0, y: 0 }, b: { x: 4000, y: 0 }, thickness: 300, ref: 'right', phase: 'existing' },
    { id: 'w', a: { x: 0, y: 0 }, b: { x: 0, y: 3000 }, thickness: 200, ref: 'left', phase: 'existing' },
  ]
  const openings = [{ id: 'o', wallId: 'n', kind: 'window' as const, offset: 1000, width: 900, phase: 'existing' as const }]
  const v = centreLines({ walls, openings })
  const n = v.walls.find((x) => x.id === 'n')!
  // The face's a-end is at x 0; the view runs the centre line to the west wall's centre line, 100 mm further west.
  expect(n.a.x).toBe(-100)
  expect(n.a.x + v.openings![0].offset).toBe(1000)
})

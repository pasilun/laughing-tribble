Do: `manage`: the actions that create, copy, rename, tag, share, trash and restore documents. Each action is an entry of the `MANAGE_ACTIONS` registry; `needsId` says whether the call needs `id`. The arguments of an action go in `args`. Floor plans only in this card (site plans follow). `share` with no `with` or `remove` shows who has the plan; sharing needs personal keys, which the test environment does not have, so the test only checks that it answers. Do not change the old tools.
The code below was written and run by the spec author in a scratch copy of the repository (typecheck, lint and the card's tests passed). Copy it as it is; if something does not compile or a test fails, say so in your report instead of redesigning.

```ts
import { currentUser } from '../../auth/user'
import { hasChanges } from '../../domain/sheet'
import { links } from '../../mcp/format'
import { createPlan, importPlan, loadPlan, renamePlan, tagPlan } from '../../service'
import { deleteDoc, PLANS, restoreDoc, TRASH_DAYS } from '../../storage/docs'
import { addShare, participants, removeShare } from '../../storage/shares'
import { listPlansAndTrash } from '../../storage/store'
import type { Plan } from '../../domain/types'
import { answer } from '../answer'
import { badArgument, invalidOperation, notFound } from '../findings'
import { Registry, type Entry } from '../names'
import { MANAGE_ACTIONS } from '../registries'
import { toToolResult, type ToolResult } from '../result'
import type { Finding, Next } from '../types'
import type { Args } from './index'

const fail = (findings: Finding[]) => toToolResult(answer({ ok: false, findings }), { summary: findings[0].message })
void Registry
export type ManageEntry = Entry & { needsId: boolean }

export async function manage(args: Args): Promise<ToolResult> {
  if (typeof args.action !== 'string' || !args.action) return fail([badArgument('action', 'krävs')])
  if (args.id !== undefined && typeof args.id !== 'string') return fail([badArgument('id', 'ska vara en sträng')])
  if (args.args !== undefined && (typeof args.args !== 'object' || args.args === null || Array.isArray(args.args))) return fail([badArgument('args', 'ska vara ett objekt')])
  const r = MANAGE_ACTIONS.resolve(args.action)
  if ('finding' in r) return fail([r.finding])
  const a = (args.args ?? {}) as Record<string, unknown>
  const name = r.entry.name
  const id = typeof args.id === 'string' ? args.id : ''
  if ((r.entry as ManageEntry).needsId && !id) return fail([badArgument('id', `krävs för åtgärden ${name}`)])
  try {
    if (name === 'create') {
      const template = a.template === 'demo' ? 'demo' : 'empty'
      const res = await createPlan({ name: String(a.name ?? 'Planritning'), template, author: 'claude' })
      if ('errors' in res) return fail(res.errors!.map(invalidOperation))
      const next: Next[] = [{ tool: 'source', arguments: { id: res.plan.id, action: 'upload_link' }, why: 'Har användaren en ritning eller bild att rita av? Ge länken.' }]
      return toToolResult(answer({ ok: true, id: res.plan.id, revision: res.revision.n, next, data: { links: links(res.plan, res.revision.n) } }), { summary: `Skapad: ${res.plan.id}, revision ${res.revision.n}\n${links(res.plan, res.revision.n)}` })
    }
    if (name === 'import') {
      const res = await importPlan(a.plan as Partial<Plan> | undefined, String(a.name ?? ''), typeof a.copy_sources_from === 'string' ? a.copy_sources_from : undefined)
      if ('errors' in res) return fail(res.errors.map(invalidOperation))
      return toToolResult(answer({ ok: true, id: res.plan.id, revision: res.revision.n, data: { copied_sources: res.copied } }), { summary: `Importerad: ${res.plan.id}, revision ${res.revision.n}${res.copied !== null ? ` (${res.copied} källritningar kopierade)` : ''}\n${links(res.plan, res.revision.n)}` })
    }
    if (name === 'rename') {
      if (typeof a.name !== 'string' || !a.name.trim()) return fail([badArgument('args.name', 'krävs')])
      const res = await renamePlan(id, a.name)
      if (!res) return fail([notFound(id)])
      if (res.status !== 200) return fail(res.errors.map(invalidOperation))
      return toToolResult(answer({ ok: true, id, revision: res.revision.n }), { summary: `Bytte namn till "${a.name}", revision ${res.revision.n}.` })
    }
    if (name === 'tags') {
      if (!Array.isArray(a.tags)) return fail([badArgument('args.tags', 'ska vara en lista')])
      const res = await tagPlan(id, a.tags)
      if (res.status === 404) return fail([notFound(id)])
      if (res.status !== 200) return fail(res.errors.map(invalidOperation))
      return toToolResult(answer({ ok: true, id, data: { tags: res.tags } }), { summary: res.tags.length ? `Taggar: ${res.tags.join(', ')}` : 'Taggarna är borttagna.' })
    }
    if (name === 'share') {
      const who = typeof a.with === 'string' ? a.with : ''
      const out = typeof a.remove === 'string' ? a.remove : ''
      const err = who ? await addShare(PLANS, id, who) : out ? await removeShare(PLANS, id, out) : null
      if (err) return fail([invalidOperation(err)])
      if (out && out === currentUser()?.id) return toToolResult(answer({ ok: true, id }), { summary: `Du har tagit bort planen ${id} för dig. De andra har kvar den.` })
      const p = await participants(PLANS, id)
      if (!p) return fail([notFound(id)])
      const people = [p.ownerRemoved ? `${p.owner} (har tagit bort den för sig)` : `${p.owner} (ägare)`, ...p.members]
      return toToolResult(answer({ ok: true, id, data: { owner: p.owner, members: p.members, me: p.me } }), { summary: `Planen delas av: ${people.join(', ')}. Du är ${p.me}.` })
    }
    if (name === 'delete') {
      const how = await deleteDoc(PLANS, id)
      if (!how) return fail([notFound(id)])
      return toToolResult(answer({ ok: true, id, data: { result: how } }), { summary: how === 'trashed' ? `Planen ${id} ligger nu i papperskorgen och kan återställas i ${TRASH_DAYS} dagar.` : `Du har tagit bort planen ${id} för dig. De andra har kvar den.` })
    }
    if (name === 'restore') {
      if (!(await restoreDoc(PLANS, id))) return fail([notFound(id)])
      return toToolResult(answer({ ok: true, id }), { summary: `Planen ${id} är tillbaka i listan.` })
    }
    // trash
    const trash = (await listPlansAndTrash({ shared: false })).trash
    return toToolResult(answer({ ok: true, data: { items: trash.map((t) => ({ id: t.id, name: t.name, days_left: t.daysLeft, ...(t.sharedWith?.length ? { shared_with: t.sharedWith } : {}) })) } }), { summary: trash.length ? trash.map((t) => `${t.id}: ${t.name} – ${t.daysLeft} dagar kvar${t.sharedWith?.length ? `, delas med ${t.sharedWith.join(', ')}` : ''}`).join('\n') : 'Papperskorgen är tom.' })
  } catch (e) {
    return fail([invalidOperation((e as Error).message)])
  }
}
void loadPlan; void hasChanges
```

```ts
import { test, expect } from '@playwright/test'
import { connector } from './connector-helpers'

test('manage', async ({ request }) => {
  const name = `Manage test ${Date.now()}`
  const c = (await connector(request, 'manage', { action: 'create', args: { name, template: 'demo' } })).structuredContent!
  expect(c.ok).toBe(true); expect(c.revision).toBe(1); const id = c.id!
  expect(c.next![0]).toMatchObject({ tool: 'source', arguments: { id, action: 'upload_link' } })
  const rn = (await connector(request, 'manage', { action: 'rename', id, args: { name: `${name} 2` } })).structuredContent!
  expect(rn.ok).toBe(true); expect(rn.revision).toBe(2)
  const tg = (await connector(request, 'manage', { action: 'tags', id, args: { tags: ['Alfa', 'Beta'] } })).structuredContent!
  expect(tg.data!.tags).toEqual(['Alfa', 'Beta'])
  const sh = (await connector(request, 'manage', { action: 'share', id })).structuredContent!
  expect(typeof sh.ok).toBe('boolean') // without personal keys there is no sharing: an answer, not a crash
  const plan = ((await connector(request, 'get', { id, format: 'json' })).structuredContent!.data as { plan: unknown }).plan
  const imp = (await connector(request, 'manage', { action: 'import', args: { name: `${name} kopia`, plan } })).structuredContent!
  expect(imp.ok, JSON.stringify(imp)).toBe(true); expect(imp.id).not.toBe(id)
  const del = (await connector(request, 'manage', { action: 'delete', id })).structuredContent!
  expect(del.data).toEqual({ result: 'trashed' })
  const tr = (await connector(request, 'manage', { action: 'trash' })).structuredContent!
  expect((tr.data!.items as Array<{ id: string }>).some((i) => i.id === id)).toBe(true)
  expect((await connector(request, 'manage', { action: 'restore', id })).structuredContent!.ok).toBe(true)
  expect((await connector(request, 'manage', { action: 'creat' })).structuredContent!.findings![0].data!.nearest).toBe('create')
  expect((await connector(request, 'manage', { action: 'rename', args: { name: 'x' } })).structuredContent!.findings![0].code).toBe('bad-argument')
  expect((await connector(request, 'manage', { action: 'rename', id: 'no-such-plan', args: { name: 'x' } })).structuredContent!.findings![0].code).toBe('not-found')
  expect((await connector(request, 'manage', { action: 'tags', id, args: { tags: 'x' } })).structuredContent!.findings![0].code).toBe('bad-argument')
  expect((await connector(request, 'manage', { action: 'create', id: 5 })).structuredContent!.findings![0].code).toBe('bad-argument')
})
```

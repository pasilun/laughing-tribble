Do: `check`: everything that is wrong or unfinished in a document, and the next call to make. A check **scope** is an entry of the `CHECKS` registry with a `run` function; this card builds the handler and the first scope, `permit` (what an application for interior work needs to show: `checkPlan`). The scopes `review`, `finish`, `audit` and `compare` come in the next cards. Do not change the old tools.
The code below was written and run by the spec author in a scratch copy of the repository (typecheck, lint and the card's tests passed). Copy it as it is; if something does not compile or a test fails, say so in your report instead of redesigning.

2. `lib/plan/connector/handlers/check.ts` (new). It imports `auditPlan` and `sourceIdOf` for the audit count that the later scope cards use:
```ts
import { auditPlan, sourceIdOf } from '../../assist/audit'
import { loadPlan } from '../../service'
import { answer } from '../answer'
import { badArgument, notFound } from '../findings'
import { CHECKS } from '../registries'
import type { CheckEntry } from '../checks'
import { nextCalls } from '../next'
import { toToolResult, type ToolResult } from '../result'
import type { Finding, Next } from '../types'
import type { Args } from './index'

const fail = (findings: Finding[]) => toToolResult(answer({ ok: false, findings }), { summary: findings[0].message })

export async function check(args: Args): Promise<ToolResult> {
  const id = args.id
  if (typeof id !== 'string' || !id) return fail([badArgument('id', 'krävs')])
  if (args.scope !== undefined && (!Array.isArray(args.scope) || args.scope.some((s) => typeof s !== 'string'))) return fail([badArgument('scope', 'ska vara en lista av strängar')])
  if (args.against !== undefined && (typeof args.against !== 'object' || args.against === null || Array.isArray(args.against))) return fail([badArgument('against', 'ska vara ett objekt')])
  const scopes = (args.scope as string[] | undefined)?.length ? (args.scope as string[]) : ['permit']
  const entries: CheckEntry[] = []
  const unknown: Finding[] = []
  for (const s of scopes) {
    const r = CHECKS.resolve(s)
    if ('finding' in r) unknown.push(r.finding)
    else entries.push(r.entry as CheckEntry)
  }
  if (unknown.length) return fail(unknown)
  const rev = await loadPlan(id)
  if (!rev) return fail([notFound(id)])
  const against = (args.against ?? {}) as Record<string, unknown>
  const findings: Finding[] = []
  const texts: string[] = []
  for (const e of entries) {
    const out = await e.run({ id, plan: rev.plan, revision: rev.meta.n, against })
    findings.push(...out.findings)
    if (out.text) texts.push(out.text)
  }
  // Where the plan stands: the audit count decides whether to compare with the drawing again.
  const p = rev.plan
  const sources = (p.sources ?? []).filter((x) => x.view !== 'elevation')
  let open: number | undefined
  if (scopes.includes('audit')) open = findings.filter((f) => f.code.startsWith('audit-') && f.level !== 'info').length
  else if (p.walls.length && p.openings.length && sources.some((x) => x.calibration) && sourceIdOf(p)) open = (await auditPlan(id, p)).filter((i) => i.level !== 'info').length
  const steps = nextCalls(p, rev.meta.n, { openIssues: open })
  const next: Next[] = []
  const fixes = findings.filter((f) => f.fix?.length)
  if (fixes.length) next.push({ tool: 'edit', arguments: { id, base_revision: rev.meta.n, ops: fixes.flatMap((f) => f.fix!) }, why: `Rätta ${fixes.length} anmärkningar direkt (${fixes.slice(0, 3).map((f) => f.id).join(', ')}${fixes.length > 3 ? ', …' : ''}).` })
  next.push(...steps.next)
  const summary = [`Revision ${rev.meta.n}. Klart hittills: ${steps.done.join(', ') || 'inget'}.`, ...texts, steps.note ?? ''].filter(Boolean).join('\n')
  return toToolResult(answer({ ok: true, id, revision: rev.meta.n, findings, next, data: { scopes, done: steps.done } }), { summary })
}
```

Behaviour to know: `ok` is true whenever the call itself worked, even when the findings are errors; `next` first offers one `edit` that applies every `fix` the findings carry (none from `permit`), then the calls from `nextCalls`.

import { test, expect } from '@playwright/test'
import { connector, legacy, makePlan } from './connector-helpers'

test('check: permit is the default scope; errors are findings', async ({ request }) => {
  const a = await makePlan(request, 'Check test')
  const d = (await connector(request, 'check', { id: a.id })).structuredContent!
  expect(d.ok).toBe(true); expect(d.revision).toBe(1); expect((d.data as { scopes: string[] }).scopes).toEqual(['permit'])
  const bad = (await connector(request, 'check', { id: a.id, scope: ['permt'] })).structuredContent!
  expect(bad.ok).toBe(false); expect(bad.findings![0].data!.nearest).toBe('permit')
  expect((await connector(request, 'check', { id: 'no-such-plan' })).structuredContent!.findings![0].code).toBe('not-found')
  expect((await connector(request, 'check', { id: a.id, scope: 'permit' })).structuredContent!.findings![0].code).toBe('bad-argument')
  expect((await connector(request, 'check', { id: a.id, against: [] })).structuredContent!.findings![0].code).toBe('bad-argument')
  const created = await legacy(request, 'plan_create', { name: `Check tom ${Date.now()}`, template: 'empty' })
  const eid = /Skapad: ([a-z0-9-]+)/.exec(created.content[0].text!)![1]
  const e = (await connector(request, 'check', { id: eid })).structuredContent!
  expect(e.next![0]).toMatchObject({ tool: 'source', arguments: { id: eid, action: 'upload_link' } })
})

Do: `export`: where to get the sheets. It returns links (the file routes of the app, `/api/plans/<id>/sheet?view=…&format=…&rev=…`, which need the person to be signed in), not file contents. Formats: `link` (default: editor, PDF and DXF), `pdf`, `png`, `svg`, `dxf`; `options.sheet` is `both`, `efter` or `fore`. This card adds a new family of names, `exports`, to the registries. Do not change the route or `links()`.
The code below was written and run by the spec author in a scratch copy of the repository (typecheck, lint and the card's tests passed). Copy it as it is; if something does not compile or a test fails, say so in your report instead of redesigning.

1. `lib/plan/connector/handlers/export.ts` (new):
```ts
import { hasChanges } from '../../domain/sheet'
import { host } from '../../mcp/format'
import { loadPlan } from '../../service'
import { answer } from '../answer'
import { badArgument, notFound } from '../findings'
import { Registry, type Entry } from '../names'
import { EXPORTS } from '../registries'
import { toToolResult, type ToolResult } from '../result'
import type { Finding } from '../types'
import type { Args } from './index'

const fail = (findings: Finding[]) => toToolResult(answer({ ok: false, findings }), { summary: findings[0].message })
const SHEETS = new Registry<Entry>('sheets', 'blad', [{ name: 'both', summary: 'Före and efter' }, { name: 'efter', summary: 'The sheet with changes' }, { name: 'fore', summary: 'The sheet as it is today' }])

export async function exportDoc(args: Args): Promise<ToolResult> {
  const id = args.id
  if (typeof id !== 'string' || !id) return fail([badArgument('id', 'krävs')])
  if (typeof args.format !== 'string' || !args.format) return fail([badArgument('format', 'krävs')])
  if (args.options !== undefined && (typeof args.options !== 'object' || args.options === null || Array.isArray(args.options))) return fail([badArgument('options', 'ska vara ett objekt')])
  const f = EXPORTS.resolve(args.format)
  if ('finding' in f) return fail([f.finding])
  const options = (args.options ?? {}) as Record<string, unknown>
  const rev = await loadPlan(id)
  if (!rev) return fail([notFound(id)])
  let sheet: string | undefined
  if (options.sheet !== undefined) {
    const s = typeof options.sheet === 'string' ? SHEETS.resolve(options.sheet) : null
    if (!s) return fail([badArgument('options.sheet', 'ska vara en sträng')])
    if ('finding' in s) return fail([s.finding])
    sheet = s.entry.name
  }
  const h = host()
  const n = rev.meta.n
  const url = (format: string, view: string) => `${h}/api/plans/${id}/sheet?view=${view}&format=${format}&rev=${n}`
  const changes = hasChanges(rev.plan)
  const links: Record<string, string> = { editor: `${h}/plans/${id}` }
  const name = f.entry.name
  if (name === 'link') {
    links.pdf = url('pdf', sheet ?? (changes ? 'both' : 'efter'))
    links.dxf = url('dxf', sheet === 'both' ? 'efter' : (sheet ?? 'efter'))
  } else if (name === 'pdf') links.pdf = url('pdf', sheet ?? (changes ? 'both' : 'efter'))
  else links[name] = url(name, sheet === 'both' ? 'efter' : (sheet ?? 'efter'))
  const lines = Object.entries(links).map(([k, v]) => `${k === 'editor' ? 'Editor' : k.toUpperCase()}: ${v}`)
  return toToolResult(answer({ ok: true, id, revision: n, data: { links } }), { summary: `Revision ${n}. Länkarna kräver inloggning.\n${lines.join('\n')}` })
}
```

```ts
import { test, expect } from '@playwright/test'
import { connector, makePlan } from './connector-helpers'

test('export', async ({ request }) => {
  const { id } = await makePlan(request, 'Export test')
  const l = (await connector(request, 'export', { id, format: 'link' })).structuredContent!
  const links = (l.data as { links: Record<string, string> }).links
  expect(l.ok).toBe(true); expect(l.revision).toBe(1)
  expect(links.editor).toContain(`/plans/${id}`); expect(links.pdf).toContain('format=pdf'); expect(links.dxf).toContain('format=dxf')
  const png = (await connector(request, 'export', { id, format: 'png', options: { sheet: 'fore' } })).structuredContent!
  expect((png.data as { links: Record<string, string> }).links.png).toContain('view=fore&format=png')
  const pdf = (await connector(request, 'export', { id, format: 'pdf', options: { sheet: 'both' } })).structuredContent!
  expect((pdf.data as { links: Record<string, string> }).links.pdf).toContain('view=both&format=pdf')
  expect((await connector(request, 'export', { id, format: 'pdff' })).structuredContent!.findings![0].data!.nearest).toBe('pdf')
  expect((await connector(request, 'export', { id, format: 'pdf', options: { sheet: 'bth' } })).structuredContent!.findings![0].data!.nearest).toBe('both')
  expect((await connector(request, 'export', { id: 'no-such-plan', format: 'pdf' })).structuredContent!.findings![0].code).toBe('not-found')
  expect((await connector(request, 'export', { id })).structuredContent!.findings![0].code).toBe('bad-argument')
})
```

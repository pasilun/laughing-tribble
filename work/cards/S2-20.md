Do: `view`: one picture. Views: `sheet` (the drawing sheet; `options.sheet` is `efter`, the default, or `fore`), `overlay` (the plan drawn over the source drawing) and `source` (the source alone, with a grid in mm). `region` zooms the last two: `{ id }` of a wall, opening or room, or `{ rect_mm: [x, y, w, h] }` in plan mm; without it the whole drawing. `options.source` picks a source. A plan without a source answers `no-source` with a `next` to `source`/`upload_link`. `max_tokens` is accepted and not used yet. Do not change `plan_view`.

import { sourceIdOf } from '../../assist/audit'
import { wallQuad } from '../../domain/geometry'
import type { Plan, View } from '../../domain/types'
import { preview } from '../../mcp/format'
import { resolveSource } from '../../scan/raster'
import { scanView } from '../../scan/view'
import { loadPlan } from '../../service'
import { answer } from '../answer'
import { badArgument, notFound } from '../findings'
import { Registry, type Entry } from '../names'
import { VIEWS } from '../registries'
import { toToolResult, type ImageBlock, type ToolResult } from '../result'
import { finding } from '../answer'
import type { Finding } from '../types'
import type { Args } from './index'

type Rect = { x: number; y: number; w: number; h: number }
const SHEETS = new Registry<Entry>('sheets', 'blad', [{ name: 'efter', summary: 'The sheet with changes (default)' }, { name: 'fore', summary: 'The sheet as it is today' }])
const fail = (findings: Finding[]) => toToolResult(answer({ ok: false, findings }), { summary: findings[0].message })

// The rectangle of an element, in plan mm, padded 500 mm: a wall, an opening (its wall) or a room (its label point).
function elementRect(plan: Plan, id: string): Rect | null {
  const pad = 500
  const around = (pts: Array<{ x: number; y: number }>): Rect => {
    const xs = pts.map((p) => p.x), ys = pts.map((p) => p.y)
    const x = Math.min(...xs) - pad, y = Math.min(...ys) - pad
    return { x, y, w: Math.max(...xs) - Math.min(...xs) + 2 * pad, h: Math.max(...ys) - Math.min(...ys) + 2 * pad }
  }
  const w = plan.walls.find((x) => x.id === id)
  if (w) return around(wallQuad(w))
  const o = plan.openings.find((x) => x.id === id)
  const ow = o && plan.walls.find((x) => x.id === o.wallId)
  if (ow) return around(wallQuad(ow))
  const r = plan.rooms.find((x) => x.id === id)
  if (r) return around(r.polygon?.length ? r.polygon : [r.at])
  return null
}

export async function view(args: Args): Promise<ToolResult> {
  const id = args.id
  if (typeof id !== 'string' || !id) return fail([badArgument('id', 'krävs')])
  if (args.view !== undefined && typeof args.view !== 'string') return fail([badArgument('view', 'ska vara en sträng')])
  if (args.region !== undefined && (typeof args.region !== 'object' || args.region === null)) return fail([badArgument('region', 'ska vara ett objekt')])
  if (args.options !== undefined && (typeof args.options !== 'object' || args.options === null || Array.isArray(args.options))) return fail([badArgument('options', 'ska vara ett objekt')])
  const which = VIEWS.resolve((args.view as string | undefined) ?? 'sheet')
  if ('finding' in which) return fail([which.finding])
  const options = (args.options ?? {}) as Record<string, unknown>
  const sheet = SHEETS.resolve(typeof options.sheet === 'string' ? options.sheet : 'efter')
  if ('finding' in sheet) return fail([sheet.finding])
  const sheetView = sheet.entry.name as View
  const rev = await loadPlan(id)
  if (!rev) return fail([notFound(id)])
  const plan = rev.plan
  const name = which.entry.name
  const done = (image: ImageBlock, summary: string, data?: Record<string, unknown>) => toToolResult(answer({ ok: true, id, revision: rev.meta.n, ...(data ? { data } : {}) }), { summary, images: [image] })
  if (name === 'sheet') {
    if (args.region !== undefined) return fail([badArgument('region', 'gäller bara vyerna overlay och source')])
    return done(await preview(plan, sheetView, false), `Revision ${rev.meta.n}`)
  }
  const sid = sourceIdOf(plan, typeof options.source === 'string' ? options.source : undefined)
  if (!sid) {
    const f = finding({ code: 'no-source', level: 'error', message: 'Planen har ingen källritning att visa.' })
    return toToolResult(answer({ ok: false, id, revision: rev.meta.n, findings: [f], next: [{ tool: 'source', arguments: { id, action: 'upload_link' }, why: 'Be användaren ladda upp ritningen.' }] }), { summary: f.message })
  }
  let raster
  try { raster = resolveSource(plan, sid) } catch (e) { return fail([badArgument('options', (e as Error).message)]) }
  let rect: Rect = { x: raster.x, y: raster.y, w: raster.widthPx * raster.mmPerPx, h: raster.heightPx * raster.mmPerPx }
  const region = args.region as { id?: unknown; rect_mm?: unknown } | undefined
  if (region) {
    if (typeof region.id === 'string') {
      const r = elementRect(plan, region.id)
      if (!r) return fail([finding({ code: 'unknown-element', level: 'error', refs: [region.id], message: `Elementet ${region.id} finns inte.` })])
      rect = r
    } else if (Array.isArray(region.rect_mm) && region.rect_mm.length === 4 && region.rect_mm.every((n) => typeof n === 'number') && (region.rect_mm[2] as number) > 0 && (region.rect_mm[3] as number) > 0) {
      const [x, y, w, h] = region.rect_mm as number[]
      rect = { x, y, w, h }
    } else return fail([badArgument('region', 'ge id eller rect_mm ([x, y, w, h] i mm)')])
  }
  try {
    const v = await scanView(id, plan, raster, rect, { overlay: name === 'overlay', view: sheetView, maxPx: 1400 })
    const img: ImageBlock = { type: 'image', data: v.png.toString('base64'), mimeType: 'image/png' }
    return done(img, `Revision ${rev.meta.n}, ${name === 'overlay' ? 'planen över ritningen' : 'ritningen'}; rutnätet är i mm.`, { rect_mm: [rect.x, rect.y, rect.w, rect.h], source: sid })
  } catch (e) {
    return fail([badArgument('region', (e as Error).message)])
  }
}

import { test, expect } from '@playwright/test'
import { readFileSync } from 'fs'
import { connector, legacy, makePlan } from './connector-helpers'

test('view', async ({ request }) => {
  const { id } = await makePlan(request, 'View test')
  const sheet = await connector(request, 'view', { id })
  expect(sheet.structuredContent!.ok).toBe(true); expect(sheet.content[1].type).toBe('image')
  expect((await connector(request, 'view', { id, options: { sheet: 'fore' } })).content[1].type).toBe('image')
  expect((await connector(request, 'view', { id, view: 'sheet', region: { rect_mm: [0, 0, 100, 100] } })).structuredContent!.findings![0].code).toBe('bad-argument')
  expect((await connector(request, 'view', { id, view: 'sheat' })).structuredContent!.findings![0].data!.nearest).toBe('sheet')
  expect((await connector(request, 'view', { id, options: { sheet: 'efer' } })).structuredContent!.findings![0].data!.nearest).toBe('efter')
  const none = (await connector(request, 'view', { id, view: 'overlay' })).structuredContent!
  expect(none.ok).toBe(false); expect(none.findings![0].code).toBe('no-source'); expect(none.next![0]).toMatchObject({ tool: 'source', arguments: { id, action: 'upload_link' } })
  expect((await connector(request, 'view', { id: 'no-such-plan' })).structuredContent!.findings![0].code).toBe('not-found')

  const added = await legacy(request, 'plan_source_add', { plan_id: id, base_revision: 1, id: 'arkiv', filename: 'test-scan.tif', data_base64: readFileSync('e2e/fixtures/test-scan.tif').toString('base64'), mm_per_px: 10 })
  expect(added.isError, added.content[0].text).toBeFalsy()
  await legacy(request, 'plan_edit', { plan_id: id, base_revision: 2, message: 'v', ops: [{ op: 'add', kind: 'wall', item: { id: 'w1', a: { x: 500, y: 500 }, b: { x: 2500, y: 500 }, thickness: 100, phase: 'existing' } }] })
  const ov = await connector(request, 'view', { id, view: 'overlay', region: { rect_mm: [0, 0, 3000, 2000] } })
  expect(ov.structuredContent!.ok, JSON.stringify(ov.structuredContent)).toBe(true); expect(ov.content[1].type).toBe('image')
  expect(ov.structuredContent!.data).toMatchObject({ rect_mm: [0, 0, 3000, 2000], source: 'arkiv' })
  const src = await connector(request, 'view', { id, view: 'source' })
  expect(src.structuredContent!.ok).toBe(true); expect(src.content[1].type).toBe('image')
  const el = await connector(request, 'view', { id, view: 'overlay', region: { id: 'w1' } })
  expect(el.structuredContent!.ok, JSON.stringify(el.structuredContent)).toBe(true)
  expect((await connector(request, 'view', { id, view: 'overlay', region: { id: 'nope' } })).structuredContent!.findings![0].code).toBe('unknown-element')
  expect((await connector(request, 'view', { id, view: 'overlay', region: { rect_mm: [0, 0] } })).structuredContent!.findings![0].code).toBe('bad-argument')
})
